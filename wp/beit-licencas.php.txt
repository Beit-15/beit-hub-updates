final class Beit_Licencas {

    const NS        = 'beit/v1';
    const TOKEN_TTL = 7 * DAY_IN_SECONDS;   // licença vale offline por 7 dias
    const CODE_TTL  = 15 * MINUTE_IN_SECONDS;
    const KEYS      = ['work', 'finance', 'pass'];

        add_action('rest_api_init', [__CLASS__, 'routes']);

    /* ---------------- instalação ---------------- */

    public static function activate() {
        self::create_tables();
        if (!get_option('beit_lic_sk')) {
            $kp = sodium_crypto_sign_keypair();
            update_option('beit_lic_sk', base64_encode(sodium_crypto_sign_secretkey($kp)), false);
            update_option('beit_lic_pk', base64_encode(sodium_crypto_sign_publickey($kp)), false);
        }
        add_option('beit_lic_products', ['8723035' => 'work', '8723058' => 'finance', '8835091' => 'pass']);
        add_option('beit_lic_admins', get_option('admin_email'));
        add_option('beit_lic_max_devices', 2);
        add_option('beit_lic_checkout', ['work' => '', 'finance' => '', 'pass' => '']);
        add_option('beit_lic_support', get_option('admin_email'));
        if (!get_option('beit_lic_hottok')) update_option('beit_lic_learn_until', time() + 2 * HOUR_IN_SECONDS, false);
        update_option('beit_lic_version', self::VERSION);
    }

    public static function maybe_upgrade() {
        if (get_option('beit_lic_version') !== self::VERSION) {
            // 1.2.0: Beit Pass passa a ser o produto 8835091 (assinatura R$ 49,90/mês) e ganha o link oficial de checkout.
            $products = (array) get_option('beit_lic_products', []);
            foreach ($products as $id => $k) if ($k === 'pass') unset($products[$id]);
            $products['8835091'] = 'pass';
            update_option('beit_lic_products', $products);
            $checkout = (array) get_option('beit_lic_checkout', []);
            $checkout['pass'] = 'https://pay.hotmart.com/A108246551Q?off=jq25f6ub';
            $checkout['metodo'] = 'https://pay.hotmart.com/D108246674K';
            update_option('beit_lic_checkout', $checkout);
            self::activate();
        }
    }

    private static function create_tables() {
        global $wpdb;
        require_once ABSPATH . 'wp-admin/includes/upgrade.php';
        $c = $wpdb->get_charset_collate();
        dbDelta("CREATE TABLE {$wpdb->prefix}beit_lic (
            id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
            email VARCHAR(190) NOT NULL,
            product VARCHAR(20) NOT NULL,
            status VARCHAR(20) NOT NULL DEFAULT 'ativa',
            active_until BIGINT NOT NULL DEFAULT 0,
            source VARCHAR(20) NOT NULL DEFAULT 'hotmart',
            last_event VARCHAR(60) NOT NULL DEFAULT '',
            transaction_id VARCHAR(80) NOT NULL DEFAULT '',
            updated_at DATETIME NOT NULL,
            PRIMARY KEY  (id),
            UNIQUE KEY email_product (email, product)
        ) $c;");
        dbDelta("CREATE TABLE {$wpdb->prefix}beit_dev (
            id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
            email VARCHAR(190) NOT NULL,
            device VARCHAR(80) NOT NULL,
            name VARCHAR(120) NOT NULL DEFAULT '',
            last_seen DATETIME NOT NULL,
            PRIMARY KEY  (id),
            UNIQUE KEY email_device (email, device)
        ) $c;");
        dbDelta("CREATE TABLE {$wpdb->prefix}beit_log (
            id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
            created_at DATETIME NOT NULL,
            event VARCHAR(60) NOT NULL,
            email VARCHAR(190) NOT NULL DEFAULT '',
            detail TEXT,
            PRIMARY KEY  (id)
        ) $c;");
    }

    /* ---------------- utilidades ---------------- */

    private static function log($event, $email = '', $detail = '') {
        global $wpdb;
        $wpdb->insert("{$wpdb->prefix}beit_log", [
            'created_at' => current_time('mysql', true), 'event' => $event,
            'email' => $email, 'detail' => is_string($detail) ? $detail : wp_json_encode($detail),
        ]);
    }

    private static function norm_email($e) { return strtolower(trim((string) $e)); }

    private static function b64u($s) { return rtrim(strtr(base64_encode($s), '+/', '-_'), '='); }
    private static function b64u_dec($s) { return base64_decode(strtr($s, '-_', '+/') . str_repeat('=', (4 - strlen($s) % 4) % 4)); }

    private static function is_admin_email($email) {
        $list = array_filter(array_map([__CLASS__, 'norm_email'], preg_split('/[\s,;]+/', (string) get_option('beit_lic_admins', ''))));
        return in_array($email, $list, true);
    }

    /** Lista do que o email pode usar agora. */
    public static function entitlements($email) {
        global $wpdb;
        if (self::is_admin_email($email)) return self::KEYS;
        $rows = $wpdb->get_col($wpdb->prepare(
            "SELECT product FROM {$wpdb->prefix}beit_lic WHERE email = %s AND active_until > %d",
            $email, time()
        ));
        $ent = array_values(array_intersect(self::KEYS, $rows));
        if (in_array('pass', $ent, true)) return self::KEYS;
        return $ent;
    }

    private static function sign(array $payload) {
        $sk  = base64_decode(get_option('beit_lic_sk'));
        $msg = self::b64u(wp_json_encode($payload));
        return $msg . '.' . self::b64u(sodium_crypto_sign_detached($msg, $sk));
    }

    private static function read_token($token) {
        $parts = explode('.', (string) $token);
        if (count($parts) !== 2) return null;
        $pk = base64_decode(get_option('beit_lic_pk'));
        $sig = self::b64u_dec($parts[1]);
        if (strlen($sig) !== SODIUM_CRYPTO_SIGN_BYTES) return null;
        if (!sodium_crypto_sign_verify_detached($sig, $parts[0], $pk)) return null;
        $p = json_decode(self::b64u_dec($parts[0]), true);
        return is_array($p) ? $p : null;
    }

    private static function issue($email, $device) {
        $now = time();
        $ent = self::entitlements($email);
        $payload = ['e' => $email, 'ent' => $ent, 'd' => $device, 'iat' => $now, 'exp' => $now + self::TOKEN_TTL];
        return ['ok' => true, 'token' => self::sign($payload), 'email' => $email, 'ent' => $ent, 'exp' => $payload['exp']];
    }

    private static function fail($msg, $code = 400) {
        return new WP_REST_Response(['ok' => false, 'msg' => $msg], $code);
    }

    /* Notificações da Hotmart (webhook 2.0) */
    public static function webhook(WP_REST_Request $req) {
        $body = $req->get_json_params();
        if (!is_array($body)) $body = $req->get_body_params();
        $hottok = $req->get_header('x_hotmart_hottok') ?: ($body['hottok'] ?? '');
        $expected = (string) get_option('beit_lic_hottok', '');
        // Sem hottok salvo: aprende o da primeira notificação da Hotmart, só dentro da janela liberada pelo admin.
        if ($expected === '' && $hottok !== '' && time() < (int) get_option('beit_lic_learn_until', 0)) {
            update_option('beit_lic_hottok', sanitize_text_field($hottok), false);
            delete_option('beit_lic_learn_until');
            $expected = (string) $hottok;
            self::log('hottok_aprendido', '', 'hottok salvo a partir da notificação da Hotmart');
        }
        if ($expected === '' || !hash_equals($expected, (string) $hottok)) {
            self::log('webhook_negado', '', 'hottok inválido');
            return self::fail('hottok inválido', 401);
        }
        $event = strtoupper((string) ($body['event'] ?? ''));
        $data  = $body['data'] ?? [];
        $email = self::norm_email($data['buyer']['email'] ?? ($data['subscriber']['email'] ?? ''));
        $pid   = (string) ($data['product']['id'] ?? '');
        $map   = (array) get_option('beit_lic_products', []);
        $key   = $map[$pid] ?? '';
        $tx    = (string) ($data['purchase']['transaction'] ?? '');

        if (!$email || !$key) {
            self::log('webhook_ignorado', $email, ['event' => $event, 'product' => $pid]);
            return ['ok' => true, 'ignored' => true];
        }

        global $wpdb;
        $t = "{$wpdb->prefix}beit_lic";
        $row = $wpdb->get_row($wpdb->prepare("SELECT * FROM $t WHERE email = %s AND product = %s", $email, $key));
        $until = $row ? (int) $row->active_until : 0;
        $status = $row ? $row->status : 'ativa';
        $now = time();

        switch ($event) {
            case 'PURCHASE_APPROVED':
            case 'PURCHASE_COMPLETE':
                $next = (int) ($data['purchase']['date_next_charge'] ?? 0);
                $next = $next > 100000000000 ? intdiv($next, 1000) : $next; // a Hotmart manda em milissegundos
                $until = max($until, $next > $now ? $next + 3 * DAY_IN_SECONDS : $now + 35 * DAY_IN_SECONDS);
                $status = 'ativa';
                break;
            case 'SUBSCRIPTION_CANCELLATION':
                $status = 'cancelada'; // mantém o acesso até o fim do período pago
                break;
            case 'PURCHASE_CANCELED':
            case 'PURCHASE_REFUNDED':
            case 'PURCHASE_CHARGEBACK':
            case 'PURCHASE_EXPIRED':
            case 'PURCHASE_PROTEST':
                $status = strtolower(str_replace('PURCHASE_', '', $event));
                $until = $now;
                break;
            default:
                self::log('webhook_evento', $email, ['event' => $event, 'product' => $key]);
                return ['ok' => true];
        }

        $wpdb->replace($t, [
            'email' => $email, 'product' => $key, 'status' => $status, 'active_until' => $until,
            'source' => 'hotmart', 'last_event' => $event, 'transaction_id' => $tx ?: ($row->transaction_id ?? ''),
            'updated_at' => current_time('mysql', true),
        ] + ($row ? ['id' => $row->id] : []));
        self::log('webhook', $email, ['event' => $event, 'product' => $key, 'until' => gmdate('c', $until)]);
        return ['ok' => true];
    }

    public static function send_code(WP_REST_Request $req) {
        $email = self::norm_email($req->get_param('email'));
        if (!is_email($email)) return self::fail('Digite um email válido.');
        $ip = sanitize_text_field($_SERVER['REMOTE_ADDR'] ?? '');
        foreach (['e' => $email, 'i' => $ip] as $k => $v) {
            $rk = 'beit_rl_' . $k . md5($v);
            $n = (int) get_transient($rk);
            if ($n >= 5) return self::fail('Muitas tentativas. Espere alguns minutos e tente de novo.', 429);
            set_transient($rk, $n + 1, HOUR_IN_SECONDS);
        }
        if (!self::entitlements($email)) {
            self::log('codigo_sem_assinatura', $email);
            return self::fail('Não encontramos assinatura ativa para este email. Use o mesmo email da compra na Hotmart.', 404);
        }
        $code = str_pad((string) random_int(0, 999999), 6, '0', STR_PAD_LEFT);
        set_transient('beit_code_' . md5($email), ['h' => wp_hash_password($code), 'tries' => 0], self::CODE_TTL);

        $html = '<div style="font-family:Segoe UI,Arial,sans-serif;background:#141414;color:#f3f3f1;padding:32px;border-radius:14px;max-width:460px">'
              . '<p style="margin:0 0 6px;color:#9b9b96">Beit Hub</p>'
              . '<h1 style="margin:0 0 18px;font-size:22px">Seu código de acesso</h1>'
              . '<p style="font-size:34px;letter-spacing:8px;font-weight:700;color:#2B85F5;margin:0 0 18px">' . esc_html($code) . '</p>'
              . '<p style="color:#9b9b96;margin:0">Digite este código no Beit Hub. Ele vale por 15 minutos. Se não foi você que pediu, pode ignorar este email.</p></div>';
        $sent = wp_mail($email, "Seu código do Beit Hub: $code", $html, ['Content-Type: text/html; charset=UTF-8']);
        self::log($sent ? 'codigo_enviado' : 'codigo_falhou', $email);
        if (!$sent) return self::fail('Não consegui enviar o email agora. Tente de novo em instantes.', 500);
        return ['ok' => true, 'msg' => 'Enviamos um código para ' . $email . '.'];
    }

    public static function verify(WP_REST_Request $req) {
        global $wpdb;
        $email  = self::norm_email($req->get_param('email'));
        $code   = preg_replace('/\D/', '', (string) $req->get_param('code'));
        $device = substr(preg_replace('/[^a-f0-9]/', '', strtolower((string) $req->get_param('device'))), 0, 64);
        $name   = sanitize_text_field((string) $req->get_param('name'));
        if (!$email || strlen($code) !== 6 || strlen($device) < 16) return self::fail('Dados incompletos.');

        $key = 'beit_code_' . md5($email);
        $c = get_transient($key);
        if (!$c) return self::fail('O código expirou. Peça um novo.', 410);
        if ($c['tries'] >= 5) { delete_transient($key); return self::fail('Muitas tentativas. Peça um novo código.', 429); }
        if (!wp_check_password($code, $c['h'])) {
            $c['tries']++; set_transient($key, $c, self::CODE_TTL);
            return self::fail('Código incorreto. Confira o email e tente de novo.', 401);
        }
        delete_transient($key);

        $t = "{$wpdb->prefix}beit_dev";
        $known = $wpdb->get_var($wpdb->prepare("SELECT id FROM $t WHERE email = %s AND device = %s", $email, $device));
        if (!$known && !self::is_admin_email($email)) {
            $count = (int) $wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM $t WHERE email = %s", $email));
            $max = max(1, (int) get_option('beit_lic_max_devices', 2));
            if ($count >= $max) {
                self::log('limite_computadores', $email);
                return self::fail("Esta assinatura já está ativa em $max computadores. Saia da conta em um deles pelo Beit Hub, ou fale com o suporte.", 409);
            }
        }
        $wpdb->replace($t, ['email' => $email, 'device' => $device, 'name' => $name, 'last_seen' => current_time('mysql', true)] + ($known ? ['id' => $known] : []));
        self::log('login', $email, $name);
        return self::issue($email, $device);
    }

    public static function refresh(WP_REST_Request $req) {
        global $wpdb;
        $p = self::read_token($req->get_param('token'));
        if (!$p) return self::fail('Licença inválida.', 401);
        $t = "{$wpdb->prefix}beit_dev";
        $id = $wpdb->get_var($wpdb->prepare("SELECT id FROM $t WHERE email = %s AND device = %s", $p['e'], $p['d']));
        if (!$id) return self::fail('Este computador saiu da conta. Entre de novo.', 401);
        $wpdb->update($t, ['last_seen' => current_time('mysql', true)], ['id' => $id]);
        return self::issue($p['e'], $p['d']);
    }

    public static function logout(WP_REST_Request $req) {
        global $wpdb;
        $p = self::read_token($req->get_param('token'));
        if ($p) $wpdb->delete("{$wpdb->prefix}beit_dev", ['email' => $p['e'], 'device' => $p['d']]);
        return ['ok' => true];
    }

    public static function save_settings() {
        if (!current_user_can('manage_options')) wp_die('Sem permissão');
        check_admin_referer('beit_lic_save');
        update_option('beit_lic_hottok', sanitize_text_field(wp_unslash($_POST['hottok'] ?? '')));
        update_option('beit_lic_admins', sanitize_textarea_field(wp_unslash($_POST['admins'] ?? '')));
        update_option('beit_lic_support', sanitize_email(wp_unslash($_POST['support'] ?? '')));
        update_option('beit_lic_max_devices', max(1, (int) ($_POST['max_devices'] ?? 2)));
        $products = []; $checkout = [];
        foreach (self::KEYS as $k) {
            $id = preg_replace('/\D/', '', (string) ($_POST["pid_$k"] ?? ''));
            if ($id) $products[$id] = $k;
            $checkout[$k] = esc_url_raw(wp_unslash($_POST["checkout_$k"] ?? ''));
        }
        update_option('beit_lic_products', $products);
        update_option('beit_lic_checkout', $checkout);
        self::back('Configurações salvas.');
    }

    public static function manual_grant() {
        if (!current_user_can('manage_options')) wp_die('Sem permissão');
        check_admin_referer('beit_lic_grant');
        global $wpdb;
        $email = self::norm_email(wp_unslash($_POST['email'] ?? ''));
        $prod  = sanitize_key($_POST['product'] ?? '');
        $days  = max(1, (int) ($_POST['days'] ?? 30));
        if (!is_email($email) || !in_array($prod, self::KEYS, true)) self::back('Email ou produto inválido.');
        $t = "{$wpdb->prefix}beit_lic";
        $row = $wpdb->get_row($wpdb->prepare("SELECT id FROM $t WHERE email = %s AND product = %s", $email, $prod));
        $wpdb->replace($t, ['email' => $email, 'product' => $prod, 'status' => 'ativa', 'active_until' => time() + $days * DAY_IN_SECONDS,
            'source' => 'manual', 'last_event' => 'MANUAL', 'updated_at' => current_time('mysql', true)] + ($row ? ['id' => $row->id] : []));
        self::log('manual', $email, "$prod $days dias");
        self::back("Acesso liberado para $email.");
    }

    public static function manual_revoke() {

    }
